A retrieval stack that runs entirely on your machine. Useful for two things:
working on a train, and finding out what self-hosting actually costs in
attention before you commit to it.

**STARTER TEMPLATE — not a production deployment.** Secrets come from `.env`,
volumes are local, and the resource limits are a starting point rather than a
measurement.

## Running it

```bash
cp .env.example .env
docker compose up -d
docker compose exec ollama ollama pull llama3.1:8b
curl localhost:8080/health
```

The first pull is several gigabytes. The health checks below are tuned for that
— a model server takes minutes to load weights, and a readiness probe with
default timings restarts it forever, which looks like a crash loop and is
actually an impatient probe.

```yaml path=docker-compose.yml
    healthcheck:
      test: ["CMD", "ollama", "list"]
      interval: 30s
      timeout: 10s
      retries: 5
      start_period: 120s
    deploy:
      resources:
        reservations:
          devices:
            - driver: nvidia
              count: all
              capabilities: [gpu]

  qdrant:
    image: qdrant/qdrant:latest
    restart: unless-stopped
    ports:

      - qdrant-data:/qdrant/storage

      test: ["CMD-SHELL", "bash -c ':> /dev/tcp/127.0.0.1/6333'"]
```

```dockerfile path=Dockerfile
FROM python:3.13-slim

COPY --from=ghcr.io/astral-sh/uv:latest /uv /usr/local/bin/uv
```

```python path=app/main.py
"""Ingest and ask, against local Ollama and Qdrant."""

from __future__ import annotations

import os
import uuid

import httpx
from fastapi import FastAPI
from pydantic import BaseModel, Field

OLLAMA = os.environ.get("OLLAMA_BASE_URL", "http://ollama:11434")
QDRANT = os.environ.get("QDRANT_URL", "http://qdrant:6333")
CHAT_MODEL = os.environ.get("CHAT_MODEL", "llama3.1:8b")
EMBED_MODEL = os.environ.get("EMBED_MODEL", "nomic-embed-text")
COLLECTION = "documents"

app = FastAPI(title="Local RAG")
# A generous timeout: local generation on CPU is slow, and the default 5s
# turns a working stack into an inexplicable stream of timeouts.
client = httpx.AsyncClient(timeout=120)


@app.on_event("startup")
async def ensure_collection() -> None:
    probe = await client.post(
        f"{OLLAMA}/api/embeddings", json={"model": EMBED_MODEL, "prompt": "x"}
    )
    size = len(probe.json()["embedding"])
    # Created from a real embedding rather than a hardcoded dimension: swapping
    # the embedding model otherwise fails at insert time with a size mismatch.
    await client.put(
        f"{QDRANT}/collections/{COLLECTION}",
        json={"vectors": {"size": size, "distance": "Cosine"}},
    )


@app.get("/health")
async def health() -> dict[str, str]:
    return {"status": "ok"}


class IngestIn(BaseModel):
    text: str = Field(min_length=1)
    source: str = "untitled"


@app.post("/ingest")
async def ingest(payload: IngestIn) -> dict[str, int]:
    chunks = [part.strip() for part in payload.text.split("\n\n") if part.strip()]
    points = []
    for chunk in chunks:
        embedding = await client.post(
            f"{OLLAMA}/api/embeddings", json={"model": EMBED_MODEL, "prompt": chunk}
        )
        points.append(
            {
                "id": str(uuid.uuid4()),
                "vector": embedding.json()["embedding"],
                "payload": {"text": chunk, "source": payload.source},
            }
        )
    await client.put(f"{QDRANT}/collections/{COLLECTION}/points", json={"points": points})
    return {"chunks": len(points)}


class AskIn(BaseModel):
    question: str = Field(min_length=1)
    top_k: int = Field(default=4, ge=1, le=20)


@app.post("/ask")
async def ask(payload: AskIn) -> dict[str, object]:
    embedding = await client.post(
        f"{OLLAMA}/api/embeddings", json={"model": EMBED_MODEL, "prompt": payload.question}
    )
    found = await client.post(
        f"{QDRANT}/collections/{COLLECTION}/points/search",
        json={
            "vector": embedding.json()["embedding"],
            "limit": payload.top_k,
            "with_payload": True,
        },
    )
    hits = found.json()["result"]
    context = "\n\n".join(hit["payload"]["text"] for hit in hits)

    answer = await client.post(
        f"{OLLAMA}/api/generate",
        json={
            "model": CHAT_MODEL,
            "prompt": (
                f"Answer using only this context. Say so if it is not there.\n\n"
                f"{context}\n\nQuestion: {payload.question}"
            ),
            "stream": False,
        },
    )
    return {
        "answer": answer.json()["response"],
        "sources": [hit["payload"]["source"] for hit in hits],
    }
```

## What running this teaches you

How long a cold start takes, how much disk model weights need, and what your
laptop sounds like under sustained inference. All three are numbers you want
before you choose self-hosting, not after.
